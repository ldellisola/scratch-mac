#!/usr/bin/env bash
set -euo pipefail

# Back up a folder and keep the last five archives.
SRC="${1:-$HOME/notes}"
DEST="/var/backups"
STAMP=$(date +%Y-%m-%d)

backup() {
  local name="$1"
  tar -czf "$DEST/$name-$STAMP.tar.gz" "$SRC"
  echo "wrote $DEST/$name-$STAMP.tar.gz"
}

if [ ! -d "$SRC" ]; then
  echo "no such folder: $SRC" >&2
  exit 1
fi

for name in daily weekly; do
  backup "$name"
done

count=0
while read -r file; do
  count=$((count + 1))
  if [[ $file == *.tmp ]]; then
    rm -f "$file"
  fi
done < <(find "$SRC" -type f)

ls -t "$DEST" | tail -n +6 | xargs -r rm --
echo "done, $count files"
